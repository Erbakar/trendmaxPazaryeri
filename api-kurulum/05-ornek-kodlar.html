Kod Örnekleri

Üretim ortamında kullanabileceğiniz, dil bazlı kısa ve anlaşılır örnekler. Tüm örnekler HMAC-SHA256 imzalama sürecini içerir.

Örnek 1 — Ürünleri Listele

GET /api/v1/products?page=1&per_page=20 
bash# Önceden üretilmiş imza ile hızlı test
curl -s "https://panel.musteri.com/api/v1/products?page=1&per_page=20" \
     -H "Authorization: Bearer $API_KEY:$SIGNATURE:$TIMESTAMP:$NONCE" \
     -H "Content-Type: application/json" | jq .
PHP<?php
$apiKey    = getenv('OZY_API_KEY');
$apiSecret = getenv('OZY_API_SECRET');

$method    = 'GET';
$endpoint  = '/api/v1/products';
$timestamp = time();
$nonce     = bin2hex(random_bytes(8));

$signature = hash_hmac(
    'sha256',
    "{$method}\n{$endpoint}\n{$timestamp}\n{$nonce}",
    $apiSecret
);

curl_setopt_array($ch, [
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_HTTPHEADER     => [
        "Authorization: Bearer {$apiKey}:{$signature}:{$timestamp}:{$nonce}",
        'Content-Type: application/json',
    ],
]);

$data = json_decode(curl_exec($ch), true);
foreach ($data['data'] as $p) {
    echo "{$p['sku']} — {$p['name']} — {$p['price']} TL\n";
}
Node.jsimport crypto from 'crypto';

const baseUrl   = process.env.OZY_BASE_URL;
const apiKey    = process.env.OZY_API_KEY;
const apiSecret = process.env.OZY_API_SECRET;

const method    = 'GET';
const endpoint  = '/api/v1/products';
const timestamp = Math.floor(Date.now() / 1000);
const nonce     = crypto.randomBytes(8).toString('hex');

const signature = crypto
    .createHmac('sha256', apiSecret)
    .update(`${method}\n${endpoint}\n${timestamp}\n${nonce}`)
    .digest('hex');

const res = await fetch(`${baseUrl}${endpoint}?page=1&per_page=20`, {
    headers: {
        'Authorization': `Bearer ${apiKey}:${signature}:${timestamp}:${nonce}`,
        'Content-Type': 'application/json',
    },
});

const data = await res.json();
data.data.forEach(p => console.log(`${p.sku} — ${p.name} — ${p.price} TL`));
Pythonimport os, hmac, hashlib, time, secrets, requests

base_url   = os.getenv('OZY_BASE_URL')
api_key    = os.getenv('OZY_API_KEY')
api_secret = os.getenv('OZY_API_SECRET')

method    = 'GET'
endpoint  = '/api/v1/products'
timestamp = str(int(time.time()))
nonce     = secrets.token_hex(8)

message   = f'{method}\n{endpoint}\n{timestamp}\n{nonce}'
signature = hmac.new(api_secret.encode(), message.encode(), hashlib.sha256).hexdigest()

res = requests.get(
    f'{base_url}{endpoint}',
    params={'page': 1, 'per_page': 20},
    headers={
        'Authorization': f'Bearer {api_key}:{signature}:{timestamp}:{nonce}',
        'Content-Type': 'application/json',
    },
    timeout=30,
)
res.raise_for_status()

for p in res.json()['data']:
    print(f"{p['sku']} — {p['name']} — {p['price']} TL")

Örnek 4 — Cron ile Sipariş Senkronizasyonu

GET /cron/sync-orders 
bash# Linux crontab
*/5 * * * * curl -s "https://panel.musteri.com/cron/sync-orders" \
    -H "X-Cron-Token: $CRON_TOKEN" > /var/log/ozy-sync.log 2>&1
JavaScriptasync function syncOrders() {
    const res = await fetch('https://panel.musteri.com/cron/sync-orders', {
        headers: { 'X-Cron-Token': process.env.CRON_TOKEN },
    });
    const data = await res.json();
    console.log(`${data.synced} sipariş senkronize edildi (yeni: ${data.new_orders}, güncellenen: ${data.updated_orders})`);
}

syncOrders().catch(console.error);
PowerShell# Windows Task Scheduler ile kullanabilirsiniz
$headers = @{ 'X-Cron-Token' = $env:CRON_TOKEN }
$res     = Invoke-RestMethod -Uri "https://panel.musteri.com/cron/sync-orders" -Headers $headers
Write-Host "$($res.synced) sipariş senkronize edildi"

Örnek 5 — Sayfalama ile Tüm Kayıtları Çek

Büyük veri setleri için sayfalama döngüsü. Rate limit'e takılmamak için iki çağrı arasına küçük bekleme ekliyoruz.

PHP$page = 1;
$allProducts = [];

do {
    $data = $client->request('GET', '/api/v1/products', [
        'page'     => $page,
        'per_page' => 100,
    ]);

    $allProducts = array_merge($allProducts, $data['data']);
    $totalPages = $data['pagination']['total_pages'];
    $page++;
    usleep(150000); // 150ms bekle
} while ($page <= $totalPages);

echo "Toplam " . count($allProducts) . " ürün alındı.\n";
Node.jsconst all = [];
let page = 1, totalPages = 1;

do {
    const data = await client.request('GET', '/api/v1/products',
        { page, per_page: 100 });
    all.push(...data.data);
    totalPages = data.pagination.total_pages;
    page++;
    await new Promise(r => setTimeout(r, 150));
} while (page <= totalPages);

console.log(`Toplam ${all.length} ürün alındı.`);
Pythonall_products = []
page, total_pages = 1, 1

while page <= total_pages:
    data = client.request('GET', '/api/v1/products',
                          params={'page': page, 'per_page': 100})
    all_products.extend(data['data'])
    total_pages = data['pagination']['total_pages']
    page += 1
    time.sleep(0.15)

print(f"Toplam {len(all_products)} ürün alındı.")

İpucu. Büyük veri setlerinde updated_at parametresi ile sadece değişen kayıtları çekmek daha verimlidir. Uygulamanıza göre delta sync pattern'i kullanmayı düşünün.
